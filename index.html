function konversiLog() { const input = document.getElementById('rawInput').value.trim(); const tbody = document.getElementById('tableBody'); tbody.innerHTML = ''; hasilTeksUntukDisalin = ""; if (!input) return; let baris = input.split('\n'); baris.reverse(); let barisDataDisalin = []; // Ditambahkan ALLOBANK, ALLO BANK, BANK ALLO, dll. const bankKeywords = [ 'BCA', 'MANDIRI', 'BRI', 'BNI', 'CIMB', 'DANAMON', 'PERMATA', 'DANA', 'OVO', 'GOPAY', 'SEABANK', 'NEO', 'SHOPEEPAY', 'BSI', 'JAGO', 'BANKJAGO', 'BANK JAGO', 'BANK BSI', 'ALLOBANK', 'ALLO BANK', 'BANK ALLO', 'ALADIN', 'BLU' ]; baris.forEach(line => { if (line.trim() === '') return; let kolom = line.split(/\t|\s{2,}/).map(item => item.trim()).filter(item => item !== ''); if (kolom.length < 4) { kolom = line.split(/\s+/).map(item => item.trim()).filter(item => item !== ''); } if (kolom.length >= 3) { let waktu = "00:00:00"; let userId = "-"; let tujuanBank = "-"; let creditStatus = ""; let jumlah = "0"; let waktuDitemukan = kolom.find(k => k.includes(':')); if (waktuDitemukan) { let matchWaktu = waktuDitemukan.match(/\d{2}:\d{2}(:\d{2})?/); if (matchWaktu) waktu = matchWaktu[0]; } for (let i = kolom.length - 1; i >= 0; i--) { let nilaiBersih = kolom[i].replace(/[,.-]/g, '').trim(); if (nilaiBersih !== '' && !isNaN(nilaiBersih) && nilaiBersih.length >= 3) { jumlah = kolom[i]; break; } } let kolomBankUtuh = kolom.find(k => { let textUpper = k.toUpperCase(); return bankKeywords.some(kw => textUpper.includes(kw)); }); if (kolomBankUtuh) { tujuanBank = kolomBankUtuh; } let kandidatUser = kolom.filter(k => { let text = k.toUpperCase(); let isTime = text.includes(':') || text.match(/\d{4}-\d{2}-\d{2}/); let isAmount = (k === jumlah); let isBankFull = (k === tujuanBank); let isNumeric = !isNaN(k.replace(/[,.-]/g, '')); let isTypeOrStatus = ['WITHDRAW', 'DEPOSIT', 'ACCEPT', 'REJECT', 'PENDING', '-'].includes(text); return !isTime && !isAmount && !isBankFull && !isNumeric && !isTypeOrStatus; }); if (kandidatUser.length > 0) { userId = kandidatUser[0]; } else { userId = kolom[2] || kolom[1] || "-"; } const tr = document.createElement('tr'); tr.innerHTML = `${waktu}${userId}${tujuanBank}${creditStatus}${jumlah}`; tbody.appendChild(tr); barisDataDisalin.push(`${waktu}\t${userId}\t${tujuanBank}\t\t${jumlah}`); } }); hasilTeksUntukDisalin = barisDataDisalin.join('\n'); }
