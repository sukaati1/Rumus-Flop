function salinHasil() { if (!hasilTeksUntukDisalin) { alert("Tidak ada data untuk disalin! Silakan submit log terlebih dahulu."); return; } navigator.clipboard.writeText(hasilTeksUntukDisalin).then(() => { // Opsional: Beri umpan balik ke pengguna kalau salin data berhasil const btnCopy = document.getElementById('btnCopy'); const teksAwal = btnCopy.innerText; btnCopy.innerText = "BERHASIL DISALIN!"; btnCopy.style.background = "var(--neon-cyan)"; btnCopy.style.color = "#000000"; setTimeout(() => { btnCopy.innerText = teksAwal; btnCopy.style.background = "transparent"; btnCopy.style.color = "#ffffff"; }, 1500); }).catch(err => { console.error("Gagal menyalin teks: ", err); }); }
